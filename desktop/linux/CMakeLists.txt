cmake_minimum_required(VERSION 3.24)
project(LinkoraVPNLinux VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
pkg_check_modules(NM REQUIRED IMPORTED_TARGET libnm>=1.42)
pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
find_program(AUTH_DIALOG nm-openconnect-auth-dialog REQUIRED PATHS /usr/lib/NetworkManager /usr/libexec /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/NetworkManager)
add_executable(linkora-vpn main.cpp agent.cpp)
target_compile_definitions(linkora-vpn PRIVATE LINKORA_AUTH_DIALOG="${AUTH_DIALOG}")
target_compile_options(linkora-vpn PRIVATE -Wall -Wextra -Wpedantic -Werror -Os -ffunction-sections -fdata-sections)
target_link_options(linkora-vpn PRIVATE -Wl,--gc-sections)
target_link_libraries(linkora-vpn PRIVATE PkgConfig::GTK PkgConfig::NM PkgConfig::SECRET)
install(TARGETS linkora-vpn RUNTIME DESTINATION bin)
install(FILES io.github.sck03.linkoravpn.desktop DESTINATION share/applications)
install(FILES ../../resources/app.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.github.sck03.linkoravpn.svg)
install(FILES ../../LICENSE DESTINATION share/doc/linkora-vpn RENAME copyright)
include(CTest)
if(BUILD_TESTING)
    add_executable(desktop_auth_fixture ../../tests/linux_auth_fixture.cpp)
    add_executable(desktop_agent_tests ../../tests/linux_agent_tests.cpp agent.cpp)
    target_include_directories(desktop_agent_tests PRIVATE .)
    target_compile_definitions(desktop_agent_tests PRIVATE LINKORA_AUTH_DIALOG="$<TARGET_FILE:desktop_auth_fixture>")
    target_compile_options(desktop_agent_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(desktop_agent_tests PRIVATE PkgConfig::NM PkgConfig::SECRET)
    add_dependencies(desktop_agent_tests desktop_auth_fixture)
    add_test(NAME desktop_agent COMMAND dbus-run-session -- $<TARGET_FILE:desktop_agent_tests>)
    set_tests_properties(desktop_agent PROPERTIES TIMEOUT 20 FAIL_REGULAR_EXPRESSION "must-not-appear|never-forward|synthetic-cookie")
    add_executable(desktop_protocol_tests ../../tests/linux_desktop_tests.cpp)
    target_include_directories(desktop_protocol_tests PRIVATE .)
    target_link_libraries(desktop_protocol_tests PRIVATE PkgConfig::GTK)
    target_compile_options(desktop_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME desktop_protocol COMMAND desktop_protocol_tests)
endif()

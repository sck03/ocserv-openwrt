cmake_minimum_required(VERSION 3.24)
project(LinkoraLinux VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckIPOSupported)
check_ipo_supported(RESULT LINKORA_IPO_SUPPORTED LANGUAGES CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ${LINKORA_IPO_SUPPORTED})
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24)
pkg_check_modules(NM REQUIRED IMPORTED_TARGET libnm>=1.42)
pkg_check_modules(SECRET REQUIRED IMPORTED_TARGET libsecret-1)
add_executable(linkora main.cpp agent.cpp)
target_compile_definitions(linkora PRIVATE LINKORA_VERSION="${PROJECT_VERSION}")
target_compile_options(linkora PRIVATE -Wall -Wextra -Wpedantic -Werror -Os -ffunction-sections -fdata-sections)
target_link_options(linkora PRIVATE -Wl,--gc-sections)
target_link_libraries(linkora PRIVATE PkgConfig::GTK PkgConfig::NM PkgConfig::SECRET)
install(TARGETS linkora RUNTIME DESTINATION bin)
install(FILES io.github.sck03.linkora.desktop DESTINATION share/applications)
install(FILES ../../resources/app.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.github.sck03.linkora.svg)
install(FILES ../../LICENSE DESTINATION share/doc/linkora RENAME copyright)
install(FILES ../../docs/DESKTOP.md ../../THIRD-PARTY-NOTICES.md DESTINATION share/doc/linkora)
include(CTest)
if(BUILD_TESTING)
    add_executable(desktop_auth_fixture ../../tests/linux_auth_fixture.cpp)
    add_executable(desktop_agent_tests ../../tests/linux_agent_tests.cpp agent.cpp)
    target_include_directories(desktop_agent_tests PRIVATE .)
    target_compile_definitions(desktop_agent_tests PRIVATE LINKORA_AUTH_DIALOG="$<TARGET_FILE:desktop_auth_fixture>")
    target_compile_options(desktop_agent_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(desktop_agent_tests PRIVATE PkgConfig::NM PkgConfig::SECRET)
    add_dependencies(desktop_agent_tests desktop_auth_fixture)
    add_test(NAME desktop_agent COMMAND dbus-run-session -- $<TARGET_FILE:desktop_agent_tests>)
    set_tests_properties(desktop_agent PROPERTIES TIMEOUT 20 FAIL_REGULAR_EXPRESSION "must-not-appear|never-forward|synthetic-cookie")
    add_executable(desktop_protocol_tests ../../tests/linux_desktop_tests.cpp)
    target_include_directories(desktop_protocol_tests PRIVATE .)
    target_link_libraries(desktop_protocol_tests PRIVATE PkgConfig::GTK)
    target_compile_options(desktop_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    add_test(NAME desktop_protocol COMMAND desktop_protocol_tests)
endif()

set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME linkora)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_CONTACT "Linkora <noreply@github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Linkora desktop client for OpenConnect and ocserv")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION net)
set(CPACK_DEBIAN_COMPRESSION_TYPE xz)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "network-manager (>= 1.42), network-manager-openconnect-gnome")
set(CPACK_DEBIAN_PACKAGE_CONFLICTS "linkora-vpn")
set(CPACK_DEBIAN_PACKAGE_REPLACES "linkora-vpn")
set(CPACK_STRIP_FILES ON)
include(CPack)

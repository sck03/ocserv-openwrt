cmake_minimum_required(VERSION 3.24)
project(BridgeVPN VERSION 0.5.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR MSVC)
    message(FATAL_ERROR "Use the documented MinGW Windows toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(BRIDGE_ARCH x64)
    set(WINTUN_ARCH amd64)
else()
    set(BRIDGE_ARCH x86)
    set(WINTUN_ARCH x86)
endif()
set(BRIDGE_DEPS "${CMAKE_SOURCE_DIR}/.deps/${BRIDGE_ARCH}" CACHE PATH "Verified static OpenConnect / GnuTLS prefix")
foreach(library openconnect stoken gnutls xml2 z hogweed nettle gmp)
    if(NOT EXISTS "${BRIDGE_DEPS}/lib/lib${library}.a")
        message(FATAL_ERROR "Missing ${library}: run scripts/build-dependencies.sh ${BRIDGE_ARCH} first.")
    endif()
endforeach()
include(CTest)
add_subdirectory(client)

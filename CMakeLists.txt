cmake_minimum_required(VERSION 3.24)
project(LinkoraVPN VERSION 0.6.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR MSVC)
    message(FATAL_ERROR "Use the documented MinGW Windows toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(VPN_PRODUCT_NAME "Linkora VPN")
set(VPN_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
configure_file(resources/build_info.h.in "${VPN_GENERATED_DIR}/build_info.h" @ONLY)
configure_file(resources/app.manifest.in "${VPN_GENERATED_DIR}/app.manifest" @ONLY)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(VPN_ARCH x64)
    set(WINTUN_ARCH amd64)
else()
    set(VPN_ARCH x86)
    set(WINTUN_ARCH x86)
endif()
set(VPN_DEPS "${CMAKE_SOURCE_DIR}/.deps/${VPN_ARCH}" CACHE PATH "Verified static OpenConnect / GnuTLS prefix")
foreach(library openconnect stoken gnutls xml2 z hogweed nettle gmp)
    if(NOT EXISTS "${VPN_DEPS}/lib/lib${library}.a")
        message(FATAL_ERROR "Missing ${library}: run scripts/build-dependencies.sh ${VPN_ARCH} first.")
    endif()
endforeach()
include(CTest)
add_subdirectory(client)
